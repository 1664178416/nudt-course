import os
import random
import numpy as np
from typing import Dict, Any, List

import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.distributions import Categorical
from transformers import AdamW, AutoConfig, AutoTokenizer, BertModel

# === 固定动作集合 ===
EXPLORATION_STRATEGIES = ['claim', 'analysis', 'evidences']
MODIFICATION_STRATEGIES = ['retrieve', 'rewrite', 'counterfactual', 'analysis']

# =========================================================
# MultiPolicyAC: A2C（双头因子化策略 + 值函数 + 熵正则）
# =========================================================
class MultiPolicyAC(nn.Module):
    def __init__(self, args, config, tokenizer):
        super().__init__()
        self.encoder = BertModel.from_pretrained(args.model_name_or_path, config=config)
        self.dropout = nn.Dropout(0.5)

        # Updated action spaces
        self.exploration_space = EXPLORATION_STRATEGIES
        self.modification_space = MODIFICATION_STRATEGIES

        self.exploration_classifier = nn.Linear(config.hidden_size, len(self.exploration_space))
        self.modification_classifier = nn.Linear(config.hidden_size, len(self.modification_space))

        self.value_estimator = nn.Linear(config.hidden_size, 1)

        self.tokenizer = tokenizer
        self.args = args
        self.eps = np.finfo(np.float32).eps.item()
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        self.to(self.device)

        self.optimizer = AdamW(self.parameters(), lr=args.learning_rate)

        self.log_probs: List[torch.Tensor] = []
        self.values: List[torch.Tensor] = []
        self.rewards: List[torch.Tensor] = []
        self.entropy: List[torch.Tensor] = []

    # ---------- I/O ----------
    def build_input(self, state: List[Dict[str, str]]) -> torch.Tensor:
        dial_id = []
        s = [101, 102]  # fallback
        for turn in state[::-1]:
            s = self.tokenizer.encode(f"{turn['role']}: {turn['content']}")
            if len(dial_id) + len(s) > self.args.max_seq_length:
                break
            dial_id = s[1:] + dial_id
        inp = s[:1] + dial_id
        return torch.tensor([inp], dtype=torch.long, device=self.device)

    def forward_features(self, input_ids: torch.Tensor) -> torch.Tensor:
        outputs = self.encoder(input_ids=input_ids)
        pooled_output = outputs[1]  # pooler_output
        return self.dropout(pooled_output)

    # ---------- 行动 ----------
    def select_action(self, state: List[Dict[str, str]]) -> Dict[str, str]:
        input_ids = self.build_input(state)
        features = self.forward_features(input_ids)

        # Select exploration strategy (π_e1)
        exploration_logits = self.exploration_classifier(features)
        exploration_probs = F.softmax(exploration_logits, dim=-1)
        m_exploration = Categorical(exploration_probs)
        exploration_action = m_exploration.sample()
        logp_exploration = m_exploration.log_prob(exploration_action).sum()
        ent_exploration = m_exploration.entropy().mean()

        # Select modification strategy (π_e2) based on exploration action
        modification_logits = self.modification_classifier(features)
        modification_probs = F.softmax(modification_logits, dim=-1)
        m_modification = Categorical(modification_probs)
        modification_action = m_modification.sample()
        logp_modification = m_modification.log_prob(modification_action).sum()
        ent_modification = m_modification.entropy().mean()

        # State value estimate
        state_value = self.value_estimator(features).squeeze()

        # Store log probabilities, values, and entropy for later updates
        self.log_probs.append(logp_exploration + logp_modification)
        self.values.append(state_value)
        self.entropy.append(ent_exploration + ent_modification)

        # Return both selected exploration and modification strategies
        return {
            "exploration": self.exploration_space[int(exploration_action.item())],
            "modification": self.modification_space[int(modification_action.item())]
        }

    # ---------- 更新 ----------
    def optimize_model(self, next_state_value: torch.Tensor) -> float:
        R = next_state_value.detach()
        returns: List[torch.Tensor] = []
        advantages: List[torch.Tensor] = []

        for r, v in zip(self.rewards[::-1], self.values[::-1]):
            R = r + self.args.gamma * R
            returns.insert(0, R)
            advantages.insert(0, (R - v).detach())

        if len(advantages) == 0:
            return 0.0

        returns_t = torch.stack(returns)
        adv_t = torch.stack(advantages)
        adv_t = (adv_t - adv_t.mean()) / (adv_t.std() + self.eps)

        critic_loss = F.mse_loss(torch.stack(self.values), returns_t)
        policy_loss = torch.stack([-lp * a for lp, a in zip(self.log_probs, adv_t)]).sum()
        entropy_loss = -torch.stack(self.entropy).sum()

        total_loss = policy_loss \
                   + self.args.value_loss_coef * critic_loss \
                   + self.args.entropy_coef * entropy_loss

        self.optimizer.zero_grad()
        total_loss.backward()
        self.optimizer.step()

        self.log_probs.clear()
        self.values.clear()
        self.rewards.clear()
        self.entropy.clear()

        return float(total_loss.detach().cpu().item())

    # ---------- 保存 / 加载 ----------
    def save_model(self, output_dir: str, filename: str = "agent1.pt", extra: dict = None):
        os.makedirs(output_dir, exist_ok=True)
        ckpt = {
            "model_state": self.state_dict(),
            "optimizer_state": self.optimizer.state_dict(),
            "args": dict(vars(self.args)) if hasattr(self.args, "__dict__") else self.args,
            "model_name_or_path": getattr(self.args, "model_name_or_path", None),
            "extra": extra or {},
        }
        torch.save(ckpt, os.path.join(output_dir, filename))

    def load_model(self, ckpt_path: str, strict: bool = True):
        ckpt = torch.load(ckpt_path, map_location=self.device)
        self.load_state_dict(ckpt["model_state"], strict=strict)
        if "optimizer_state" in ckpt and ckpt["optimizer_state"]:
            self.optimizer.load_state_dict(ckpt["optimizer_state"])
        return ckpt.get("extra", {})

# ===========================================
# 执行器 + 包装：Agent1
# ===========================================
def execute_module1(exploration_name: str, modification_name: str) -> str:
    return f"[Module1] exploration={exploration_name}; modification={modification_name} -> generated text."

class Agent1:
    def __init__(self, model: MultiPolicyAC):
        self.model = model
        self.device = model.device

    def act(self, state: List[Dict[str, str]]) -> Dict[str, Any]:
        action = self.model.select_action(state)
        text = execute_module1(action['exploration'], action['modification'])
        return {"action": action, "text": text}

    def give_reward(self, r: float):
        self.model.rewards.append(torch.tensor(float(r), dtype=torch.float32, device=self.device))

    def optimize(self, terminal: bool = True) -> float:
        next_v = torch.tensor(0.0, dtype=torch.float32, device=self.device) if terminal else torch.tensor(0.0, device=self.device)
        return self.model.optimize_model(next_v)

    # 包装一层，便于 game.py 调用
    def save(self, output_dir: str, filename: str = "agent1.pt", extra: dict = None):
        self.model.save_model(output_dir, filename, extra)

    def load(self, ckpt_path: str):
        return self.model.load_model(ckpt_path)

def make_agent1(args) -> Agent1:
    tokenizer = AutoTokenizer.from_pretrained(args.model_name_or_path)
    config = AutoConfig.from_pretrained(args.model_name_or_path)
    model = MultiPolicyAC(args, config, tokenizer)
    return Agent1(model)
